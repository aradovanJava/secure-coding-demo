📄 Siguran kod (Model)

@Data
public class Comment {
    @NotBlank(message = "Ime ne smije biti prazno")
    @Size(min = 2, max = 50)
    private String name;
    
    @NotBlank(message = "Komentar ne smije biti prazan")
    @Size(min = 5, max = 500)
    private String content;
}

📄 Siguran kod (Controller)

@PostMapping("/secure")
public String submitSecureComment(
    @Valid @ModelAttribute Comment comment,  // ✅ SA @Valid
    BindingResult bindingResult) {
    
    // ✅ Provjera validacijskih grešaka
    if (bindingResult.hasErrors()) {
        return "secure";
    }
    
    // ✅ HTML sanitizacija
    String safeName = htmlSanitizer.sanitize(comment.getName());
    String safeContent = htmlSanitizer.sanitize(comment.getContent());
    
    // ✅ Dodatna provjera za JavaScript protokole
    if (containsJavaScriptProtocol(safeName) || 
        containsJavaScriptProtocol(safeContent)) {
        model.addAttribute("error", "Nije dozvoljen JavaScript!");
        return "secure";
    }
    
    secureComments.add(new Comment(safeName, safeContent));
    return "redirect:/secure";
}

📄 HTML Sanitizer konfiguracija

@Bean
public PolicyFactory htmlSanitizer() {
    return new HtmlPolicyBuilder()
        .allowElements("p", "br", "strong", "em")
        .allowElements("ul", "ol", "li")
        .allowElements("a")
        .allowAttributes("href").onElements("a")
        .allowStandardUrlProtocols()
        .toFactory();  // Sve ostalo je blokirano!
}
